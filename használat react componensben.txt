import { useProductSync } from '../hooks/useProductSync';

export function ProductList() {
  const { products, loading, error, lastSync } = useProductSync(15); // 15 perces polling

  if (loading && products.length === 0) return <div>Termékek betöltése...</div>;
  if (error) return <div>Hiba: {error}</div>;

  return (
    <div>
      <h2>Terméklista</h2>
      <p>Utolsó frissítés: {lastSync ? new Date(lastSync).toLocaleTimeString() : '-'}</p>
      <ul>
        {products.map((item) => (
          <li key={item.id}>
            [{item.source}] <strong>{item.name}</strong> - {item.price}
          </li>
        ))}
      </ul>
    </div>
  );
}
